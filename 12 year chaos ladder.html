        const canvas = document.getElementById("gameCanvas");

        const ctx = canvas.getContext("2d");


        // Game State Variables

        let climbProgress = 0; // 0 to 100%

        let staminaUsed = 0;

        let totalClicks = 0;

        let gameWon = false;

        let slipMessage = "";

        let slipTimer = 0;


        // Milestones along the climb (Left & Right alternating)

        const milestones = [

            { pos: 5, side: 'left', text: "13 Years Knowing Each Other" },

            { pos: 12, side: 'right', text: "12 Years Married & Ride or Die" },

            { pos: 22, side: 'left', text: "3 Years Long Distance" },

            { pos: 32, side: 'right', text: "Chronic Health Condition" },

            { pos: 42, side: 'left', text: "Fur Baby & Baby in Heaven" },

            { pos: 52, side: 'right', text: "7 Years Waiting for Rainbow" },

            { pos: 62, side: 'left', text: "Almost Took Me Out (ICU & Coma)" },

            { pos: 72, side: 'right', text: "Kids Kicks Cancer's Butt" },

            { pos: 82, side: 'left', text: "WHO Trauma Limit Exceeded" },

            { pos: 92, side: 'right', text: "Dark Humor & Priorities Aligned" },

            { pos: 100, side: 'top', text: "FINAL APEX: One Hella Ride!" }

        ];


        function playerClimb() {

            if (gameWon) return;


            totalClicks++;

            staminaUsed += 4;


            // Random slip-back mechanic to make it harder

            let slipChance = Math.random();

            let gain = 3.5;

            

            if (slipChance < 0.25 && climbProgress > 10) {

                climbProgress -= 4; // Slip back!

                slipMessage = "⚠️ HEAVY HURDLE! SLIPPED BACK!";

                slipTimer = 30;

            } else {

                climbProgress += gain;

                slipMessage = "";

            }


            if (climbProgress >= 100) {

                climbProgress = 100;

                gameWon = true;

                document.getElementById("climbBtninnerText = "🏆 LEVEL COMPLETED!");

                document.getElementById("climbBtn").style.backgroundColor = "#e67e22";

            }


            updateHUD();

        }


        function updateHUD() {

            document.getElementById("hud-stamina").innerText = `STAMINA: ${Math.min(staminaUsed, 9999)} PTS`;

            document.getElementById("hud-score").innerText = `HEIGHT: ${Math.floor(climbProgress)}%`;

        }


        function drawGame() {

            ctx.clearRect(0, 0, canvas.width, canvas.height);


            // Draw Background Stars / Clouds

            ctx.fillStyle = "#ffffff";

            for (let i = 0; i < 40; i++) {

                let x = (i * 37) % canvas.width;

                let y = (i * 59 + Math.floor(climbProgress)) % canvas.height;

                ctx.fillRect(x, y, 2, 2);

            }


            // Draw Parallel Career Line (Lead BI & Principal Architect)

            ctx.strokeStyle = "#3498db";

            ctx.lineWidth = 4;

            ctx.setLineDash([6, 6]);

            ctx.beginPath();

            ctx.moveTo(220, 650);

            ctx.lineTo(220, 80);

            ctx.stroke();

            ctx.setLineDash([]);


            ctx.fillStyle = "#3498db";

            ctx.font = "7px 'Press Start 2P'";

            ctx.fillText("BI Engineer & Principal Architect Career Path ➔", 80, 350);


            // Draw Central Ladder

            let ladderBottomY = 620;

            let ladderTopY = 100;

            let ladderHeight = ladderBottomY - ladderTopY;

            let currentY = ladderBottomY - (climbProgress / 100) * ladderHeight;


            ctx.fillStyle = "#7f8c8d";

            for (let y = ladderTopY; y <= ladderBottomY; y += 25) {

                ctx.fillRect(380, y, 40, 8); // rungs

            }

            ctx.fillRect(370, ladderTopY, 6, ladderHeight); // left rail

            ctx.fillRect(424, ladderTopY, 6, ladderHeight); // right rail


            // Draw Milestones clearly on Left and Right

            milestones.forEach(m => {

                let mY = ladderBottomY - (m.pos / 100) * ladderHeight;

                

                // Connector line

                ctx.strokeStyle = "#f1c40f";

                ctx.lineWidth = 2;

                ctx.beginPath();

                if (m.side === 'left') {

                    ctx.moveTo(370, mY);

                    ctx.lineTo(240, mY);

                    ctx.stroke();


                    ctx.fillStyle = "rgba(0,0,0,0.8)";

                    ctx.fillRect(30, mY - 10, 210, 20);

                    ctx.strokeStyle = "#f1c40f";

                    ctx.strokeRect(30, mY - 10, 210, 20);


                    ctx.fillStyle = "#ffffff";

                    ctx.font = "6px 'Press Start 2P'";

                    ctx.fillText(m.text, 36, mY + 3);

                } else if (m.side === 'right') {

                    ctx.moveTo(430, mY);

                    ctx.lineTo(560, mY);

                    ctx.stroke();


                    ctx.fillStyle = "rgba(0,0,0,0.8)";

                    ctx.fillRect(560, mY - 10, 210, 20);

                    ctx.strokeStyle = "#f1c40f";

                    ctx.strokeRect(560, mY - 10, 210, 20);


                    ctx.fillStyle = "#ffffff";

                    ctx.font = "6px 'Press Start 2P'";

                    ctx.fillText(m.text, 566, mY + 3);

                }

            });


            // Draw Players climbing together at current progress

            ctx.fillStyle = "#e74c3c"; // Player 1 (Heart)

            ctx.fillRect(388, currentY - 15, 10, 10);

            ctx.fillStyle = "#3498db"; // Player 2 (Shield)

            ctx.fillRect(402, currentY - 15, 10, 10);


            // Slip Alert Popup

            if (slipTimer > 0) {

                slipTimer--;

                ctx.fillStyle = "#e74c3c";

                ctx.font = "8px 'Press Start 2P'";

                ctx.fillText(slipMessage, 220, currentY - 35);

            }


            // Win State Message on Canvas

            if (gameWon) {

                ctx.fillStyle = "rgba(0, 0, 0, 0.85)";

                ctx.fillRect(100, 200, 600, 300);

                ctx.strokeStyle = "#f1c40f";

                ctx.lineWidth = 4;

                ctx.strokeRect(100, 200, 600, 300);


                ctx.fillStyle = "#f1c40f";

                ctx.font = "10px 'Press Start 2P'";

                ctx.fillText("🏆 12TH ANNIVERSARY REACHED! 🏆", 130, 240);


                ctx.fillStyle = "#ffffff";

                ctx.font = "7px 'Press Start 2P'";

                ctx.fillText(`Total Clicks / Stamina Expended: ${staminaUsed} pts`, 140, 280);

                ctx.fillText("See how hard it was to climb up and what stamina it took?", 120, 320);

                ctx.fillText("It was that times infinity to do these 12 years,", 135, 350);

                ctx.fillStyle = "#f1c40f";

                ctx.fillText("but nobody I would have as Player 2 than him.", 130, 390);

            }


            requestAnimationFrame(drawGame);

        }


        // Start render loop

        drawGame();

    </script>
